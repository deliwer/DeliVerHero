### **2. app.js**

```js
const express = require('express');
const session = require('express-session');
const bodyParser = require('body-parser');
const fs = require('fs');

const app = express();
const PORT = 3000;

app.set('view engine', 'ejs');
app.use(express.static('public'));
app.use(bodyParser.urlencoded({ extended: true }));

// Session setup
app.use(session({
  secret: 'deliwer_secret_key',
  resave: false,
  saveUninitialized: true
}));

// Load members data
let members = JSON.parse(fs.readFileSync('./data/members.json'));
let events = JSON.parse(fs.readFileSync('./data/events.json'));

// Routes

// Relocate gateway (unchanged)
app.get('/relocate', (req, res) => {
  res.render('relocate');
});

// Community login/register
app.get('/community', (req, res) => {
  res.render('login', { message: null });
});

app.post('/community/login', (req, res) => {
  const { email, tier } = req.body;
  const member = members.find(m => m.email === email && m.tier === tier);
  if(member){
    req.session.user = member;
    res.redirect('/community/dashboard');
  } else {
    res.render('login', { message: 'Member not found or wrong tier.' });
  }
});

app.post('/community/register', (req, res) => {
  const { name, email, tier } = req.body;
  if(members.find(m => m.email === email)) {
    return res.render('login', { message: 'Email already registered.' });
  }
  const newMember = { name, email, tier };
  members.push(newMember);
  fs.writeFileSync('./data/members.json', JSON.stringify(members, null, 2));
  req.session.user = newMember;
  res.redirect('/community/dashboard');
});

// Dashboard
app.get('/community/dashboard', (req, res) => {
  if(!req.session.user) return res.redirect('/community');
  // Filter events based on tier
  const userEvents = events.filter(ev => ev.tiers.includes(req.session.user.tier));
  res.render('dashboard', { user: req.session.user, events: userEvents });
});

// Logout
app.get('/community/logout', (req, res) => {
  req.session.destroy();
  res.redirect('/community');
});

app.listen(PORT, () => console.log(`Server running on http://localhost:${PORT}`));
```
