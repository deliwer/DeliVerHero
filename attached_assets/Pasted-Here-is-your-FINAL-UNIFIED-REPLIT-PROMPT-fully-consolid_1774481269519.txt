## 📦 CSV PARSER

### `/lib/parser.js`

```js
import fs from "fs";
import csv from "csv-parser";

export function loadCSV(filePath) {
  return new Promise((resolve) => {
    const results = [];
    fs.createReadStream(filePath)
      .pipe(csv())
      .on("data", (data) => results.push(data))
      .on("end", () => resolve(results));
  });
}
```

## 💾 STATE HANDLER

### `/lib/state.js`

```js
import fs from "fs";

const path = "./data/state.json";

export function loadState() {
  if (!fs.existsSync(path)) return { sent: {} };
  return JSON.parse(fs.readFileSync(path));
}

export function saveState(state) {
  fs.writeFileSync(path, JSON.stringify(state, null, 2));
}
```

## 📤 SEND AGENT

### `/api/send.js`

```js
import { loadCSV } from "../lib/parser.js";
import { loadState, saveState } from "../lib/state.js";
import { generateReferral } from "../lib/referral.js";
import { sendEmail } from "../lib/mailer.js";
import fs from "fs";

export default async function handler(req, res) {
  const brokers = await loadCSV("./data/brokers.csv");
  const state = loadState();

  let count = 0;
  const batchSize = parseInt(process.env.BATCH_SIZE || "100");

  for (const broker of brokers) {
    const email = broker.Email;

    if (!email || state.sent[email]) continue;

    const { referralId, referralLink } = generateReferral(email);

    const template = fs.readFileSync("./templates/email1.html", "utf8")
      .replace(/{{Name}}/g, broker.Name || "")
      .replace(/{{referralLink}}/g, referralLink);

    await sendEmail(
      email,
      "Earn commissions with DeliWer partner program",
      template
    );

    state.sent[email] = {
      sentAt: Date.now(),
      referralId,
      followup1: false,
      followup2: false,
      followup3: false
    };

    count++;
    if (count >= batchSize) break;
  }

  saveState(state);

  res.json({ sent: count });
}
```

## 🔁 FOLLOW-UP AGENT

### `/api/followup.js`

```js
import { loadState, saveState } from "../lib/state.js";
import { sendEmail } from "../lib/mailer.js";
import fs from "fs";

export default async function handler(req, res) {
  const state = loadState();
  const now = Date.now();

  for (const email in state.sent) {
    const user = state.sent[email];
    const days = (now - user.sentAt) / (1000 * 60 * 60 * 24);

    let templateFile = null;
    let key = null;

    if (days > 2 && !user.followup1) {
      templateFile = "followup1.html";
      key = "followup1";
    } else if (days > 5 && !user.followup2) {
      templateFile = "followup2.html";
      key = "followup2";
    } else if (days > 10 && !user.followup3) {
      templateFile = "followup3.html";
      key = "followup3";
    }

    if (!templateFile) continue;

    const template = fs.readFileSync(`./templates/${templateFile}`, "utf8");

    await sendEmail(email, "Reminder: Your DeliWer referral earnings", template);

    user[key] = true;
  }

  saveState(state);

  res.json({ status: "follow-ups sent" });
}
```
