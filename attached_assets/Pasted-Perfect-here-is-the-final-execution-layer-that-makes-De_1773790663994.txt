function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("LEADS");
  var data = JSON.parse(e.postData.contents);

  sheet.appendRow([
    new Date(),
    data.name || "",
    data.phone || "",
    data.scenario || "",
    data.from || "",
    data.to || "",
    data.date || "",
    data.ref || "",
    data.page || "",
    data.value || "",
    "New"

# 💻 3. JAVASCRIPT TRACKING SNIPPET (REPLIT / VERCEL)

Add this to every page (before closing `</body>`):

```html
<script>
(function() {

  // Capture referral code
  const urlParams = new URLSearchParams(window.location.search);
  const ref = urlParams.get('ref');

  if (ref) {
    localStorage.setItem('ref', ref);
  }

  const storedRef = localStorage.getItem('ref') || "direct";

  // Function to send lead
  function sendLead(data) {
    fetch("YOUR_GOOGLE_SCRIPT_WEBHOOK_URL", {
      method: "POST",
      body: JSON.stringify(data)
    });
  }

  // Hook into CTA buttons
  document.querySelectorAll(".cta-button").forEach(btn => {
    btn.addEventListener("click", function() {

      const leadData = {
        scenario: btn.dataset.scenario || "unknown",
        page: window.location.pathname,
        ref: storedRef,
        from: document.querySelector("#fromArea")?.value || "",
        to: document.querySelector("#toArea")?.value || "",
        date: document.querySelector("#moveDate")?.value || "",
        value: ""
      };

      sendLead(leadData);

      // WhatsApp redirect
      const message = encodeURIComponent(
        "Relocation request via deliwer.com\n" +
        "Scenario: " + leadData.scenario + "\n" +
        "From: " + leadData.from + "\n" +
        "To: " + leadData.to + "\n" +
        "Date: " + leadData.date + "\n" +
        "Referral: " + leadData.ref + "\n" +
        "Page: " + leadData.page
      );

      window.open("https://wa.me/971523946311?text=" + message, "_blank");
    });
  });

})();
</script>
```
