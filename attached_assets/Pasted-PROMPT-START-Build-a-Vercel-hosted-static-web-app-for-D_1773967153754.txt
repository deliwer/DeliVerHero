4. GOOGLE APPS SCRIPT WEBHOOK

Create webhook:

function doPost(e) {
  var sheet = SpreadsheetApp.openById("SHEET_ID").getSheetByName("Leads");
  var data = JSON.parse(e.postData.contents);

  sheet.appendRow([
    new Date(),
    data.name,
    data.phone,
    data.intent,
    data.ref,
    data.utm_source,
    data.utm_campaign,
    "New",
    "",
    ""
  ]);

  return ContentService.createTextOutput("Success");
}
