  <script>
    // ---------------- Founder Data ----------------
    const founders = [
      { name:"Hassan Jawad", number:"971523946311" },
      { name:"Rubab Hassan", number:"971567148381" }
    ];
    const today = new Date().toISOString().split('T')[0];

    function getFounderKey(name) { return "streak_" + name.replace(/\s+/g,''); }

    function loadStreak(founder) {
      let streak = parseInt(localStorage.getItem(getFounderKey(founder.name))) || 0;
      let lastPosted = localStorage.getItem(getFounderKey(founder.name)+"_last") || '';
      return { streak, lastPosted };
    }

    function saveStreak(founder, streak) {
      localStorage.setItem(getFounderKey(founder.name), streak);
      localStorage.setItem(getFounderKey(founder.name)+"_last", today);
    }

    function markPosted() {
      founders.forEach(f=>{
        let data = loadStreak(f);
        if(data.lastPosted===today) return;
        data.streak++;
        saveStreak(f,data.streak);
      });
      updateDisplay();
      alert("🔥 Streaks updated for all founders! Keep building!");
    }

    function getRank(days){
      if(days>=365) return 'Relentless';
      if(days>=90) return 'Machine';
      if(days>=30) return 'Operator';
      if(days>=7) return 'Builder';
      return 'Newbie';
    }

    function getNextLevel(days){
      if(days<7) return 7;
      if(days<30) return 30;
      if(days<90) return 90;
      if(days<365) return 365;
      return 365;
    }

    function updateDisplay(){
      // Display main founder (Hassan) streak
      let data = loadStreak(founders[0]);
      document.getElementById('streak').innerText = data.streak;
      document.getElementById('badge').innerText = getRank(data.streak);
      let progressPercent = Math.min(100,(data.streak/getNextLevel(data.streak))*100);
      document.getElementById('progressFill').style.width = progressPercent+'%';

      // Update leaderboard
      founders.forEach(f=>{
        let fdata = loadStreak(f);
        if(f.name==="Hassan Jawad") document.getElementById('hassanStreak').innerText=fdata.streak;
        if(f.name==="Rubab Hassan") document.getElementById('rubabStreak').innerText=fdata.streak;

        // Show missed day alert
        if(fdata.lastPosted !== today){
          document.getElementById('alertMissedDay').style.display='block';
        } else {
          document.getElementById('alertMissedDay').style.display='none';
        }
      });
    }

    // ---------------- ChatGPT ----------------
    function openChatGPT(){
      const prompt = encodeURIComponent(
`Generate a short, powerful daily founder post for DeliWer in Dubai.
Focus: sustainability, relocation, water systems, circular economy.
Tone: visionary but grounded.
Include website www.deliwer.com and 3 relevant hashtags.`
      );
      window.open(`https://chat.openai.com/?q=${prompt}`,"_blank");
    }

    // ---------------- WhatsApp ----------------
    function pushToWhatsApp(){
      const founder = founders[0]; // Morning must-read founder
      const message = prompt("Paste your daily post here to push to WhatsApp:");
      if(message){
        const encoded = encodeURIComponent(message);
        window.open(`https://wa.me/${founder.number}?text=${encoded}`,"_blank");
      }
    }

    // ---------------- Meta Business Suite ----------------
    function openMeta(){
      alert("Open Meta Business Suite now and post FB/IG content!");
      window.open("https://business.facebook.com/","_blank");
    }

    // ---------------- Morning WhatsApp Trigger ----------------
    function scheduleMorningWhatsApp(){
      setInterval(()=>{
        let now = new Date();
        if(now.getHours()===10 && now.getMinutes()===0){
          founders.forEach(f=>{
            const msg = encodeURIComponent(
              `🔥 Daily Founder Update - Must read: Open WhatsApp and check posts for today!`
            );
            window.open(`https://wa.me/${f.number}?text=${msg}`,"_blank");
          });
          const btn=document.getElementById('whatsappBtn');
          btn.classList.add('morning');
          btn.innerText="WhatsApp - Must Read!";
        }
      },60000);
    }

    // ---------------- Initialize ----------------
    window.onload=()=>{
      updateDisplay();
      scheduleMorningWhatsApp();
    }
  </script>
